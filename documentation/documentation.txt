# Pseudocode for my classification of messages

function classify_message(message): 
   normalise message 
   if contains crisis keyword: 
       return "crisis" 
   else if contains anxiety keyword: 
       return "anxiety" 
   else if contains depression keyword: 
       return "depression" 
   else: 
       return "general"


# Pseudocode for the response generation

if crisis: 
   return supportive message + emergency signposting 
if anxiety: 
   return grounding-style reassurance 
if depression: 
   return validation + encouragement 
else: 
   return general supportive message
